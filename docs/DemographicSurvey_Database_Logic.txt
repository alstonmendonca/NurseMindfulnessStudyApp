// Function to submit demographic survey to Supabase
const handleSubmit = async (
  participantNumber: number,
  responses: DemographicSurveyData,
  otherResponses: Record<string, string>
) => {
  try {
    // First check if survey already exists for this participant
    const { data: existingSurvey, error: checkError } = await supabase
      .from('demographic_surveys')
      .select('id')
      .eq('participant_id', participantNumber)
      .single();

    if (!checkError && existingSurvey) {
      // Survey already exists - handle accordingly
      console.log('Survey already exists for this participant');
      return { success: false, error: 'Survey already exists' };
    }

    // Prepare data for submission - map camelCase to snake_case
    const surveyData = {
      participant_id: participantNumber,
      sample_code: responses.sampleCode,
      age_group: responses.ageGroup,
      gender: responses.gender,
      marital_status: responses.maritalStatus,
      educational_qualification: responses.educationalQualification,
      educational_other: otherResponses.educationalQualificationOther || null,
      designation: responses.designation,
      income_level: responses.incomeLevel,
      years_experience: responses.yearsExperience,
      working_unit: responses.workingUnit,
      working_unit_other: otherResponses.workingUnitOther || null,
      work_shift: responses.workShift,
      hours_per_day: responses.hoursPerDay,
      night_shifts_per_month: responses.nightShiftsPerMonth,
      night_shifts_other: otherResponses.nightShiftsPerMonthOther || null,
      place_of_residence: responses.placeOfResidence,
      residence_other: otherResponses.placeOfResidenceOther || null,
    };

    // Submit to database
    const { error: surveyError } = await supabase
      .from('demographic_surveys')
      .insert(surveyData);

    if (surveyError) {
      console.error('Error saving survey:', surveyError);
      return { success: false, error: surveyError.message };
    }

    // Optionally update participant record to mark survey as completed
    const { error: updateError } = await supabase
      .from('participants')
      .update({ demographic_survey_completed: true })
      .eq('participant_number', participantNumber);

    if (updateError) {
      console.error('Error updating participant:', updateError);
    }

    return { success: true };

  } catch (error) {
    console.error('Error in handleSubmit:', error);
    return { success: false, error: String(error) };
  }
};

// Check if all required questions are answered
const isComplete = (responses: Partial<DemographicSurveyData>, otherResponses: Record<string, string>): boolean => {
  return DEMOGRAPHIC_QUESTIONS.filter(q => q.required).every(question => {
    const value = responses[question.id as keyof DemographicSurveyData];
    if (!value) return false;

    // Check if "other" option is selected but no text provided
    const isOtherOption = value.toLowerCase().includes('other') || value.toLowerCase().includes('any other');
    if (isOtherOption && question.type === 'other') {
      const otherValue = otherResponses[`${question.id}Other`];
      return otherValue && otherValue.trim().length > 0;
    }

    return true;
  });
};
